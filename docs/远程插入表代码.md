/**
 * 核心业务入口 - 远程数据上报逻辑
 * 
 * 流程：
 * 1. 参数校验（contractNumber, specimenNumber, inspectionName, expId, labId, fileName）
 * 2. 获取申请单信息（apl_contract_plan），校验状态为"测试中"或"录入中"
 * 3. 检查台位是否开启自动报告
 * 4. 获取检测项信息（apl_contract_inspection）
 * 5. 构建并保存 TestData
 * 6. 更新计划状态
 * 7. 更新样机信息（试制样机 或 GTC2样机）
 * 8. 根据测试项目类型分发处理
 * 9. 文件上传与附件关联
 * 10. 计算综合结论
 */

public NewResult<String> processAutoReport(SubmitParams params, TestDataCompressor testDataCompressor, MultipartFile file) {
    // 1. 参数校验
    if (params == null || params.getContractNumber() == null || params.getSpecimenNumber() == null ||
            params.getInspectionName() == null || params.getExpId() == null ||
            params.getLabId() == null || params.getFileName() == null) {
        throw new IllegalArgumentException("请求参数不完整！");
    }

    // 2. 获取申请单信息
    AplContractPlan aplContractPlan = aplContractPlanService.selectOneByContractNumberNoStatus(params.getContractNumber());
    if (aplContractPlan == null) {
        throw new RuntimeException("未找到申请单信息！");
    }
    
    // 定义允许的申请单状态列表
    List<String> statusNames = Arrays.asList(
            "测试中",
            "录入中"
    );
    // 如果不在范围内，直接报错
    if (!statusNames.contains(aplContractPlan.getStatusName())) {
        throw new RuntimeException("当前申请单不为【测试中或者录入中】的状态，不可触发自动报告功能！");
    }
    
    // 3.判断当前台位是否选择自动报告，如果否直接返回
    boolean labifAutoReport = tsInstrumentAbilityService.selectOneLabifAutoReport(aplContractPlan.getInstrumentNumber());
    if(!labifAutoReport){
        throw new RuntimeException("当前台位的自动报告选项为否，不可触发自动报告功能！！");
    }
    
    // 4. 获取检测项信息
    AplContractInspection aplContractInspection = aplContractInspectionService.getStandardByInspectionCompressor(
        aplContractPlan.getId(), aplContractPlan.getContractType(), params.getInspectionName());

    if (aplContractInspection != null) {
        boolean ifExistOtherStatus = iTestDataService.ifExistOtherStatus(aplContractInspection.getContractinspectionid());
        if (ifExistOtherStatus) {
            throw new RuntimeException("原始记录存在其他状态数据，不可写入！");
        }
    }

    if (aplContractInspection == null) {
        aplContractInspection = aplContractInspectionService.getStandardInfo(aplContractPlan.getId(), params.getInspectionName());
    }

    if (aplContractInspection == null) {
        throw new RuntimeException("项目名称关联失败！");
    }
    
    // 5. 构建并保存 TestData
    TestData testData = buildAndSaveTestData(aplContractPlan, aplContractInspection, testDataCompressor, params.getSpecimenNumber());
    
    // 6. 更新计划状态
    aplContractPlanService.updateInfoByContractNumber(params.getContractNumber());
    
    // 7. 更新样机信息
    TsSpecimen tsSpecimen = tsSpecimenService.getSpecimenInfo(params.getSpecimenNumber());
    if (tsSpecimen != null) {
        List<AplPlanLogs> aplPlanLogs = iAplPlanLogsSevice.selectByContractID(aplContractPlan.getId());
        iTestDataSampleSevice.addOrUpdateTestDataSample(tsSpecimen, testData, aplPlanLogs);
    } else {
        // 查找GTC2样机
        TsSpecimenCheckin tsSpecimenCheckin = tsSpecimenCheckinService.getSpecimenCheckinInfo(params.getSpecimenNumber());
        if (tsSpecimenCheckin != null) {
            List<AplPlanLogs> aplPlanLogs = iAplPlanLogsSevice.selectByContractID(aplContractPlan.getId());
            tsSpecimen = new TsSpecimen();
            tsSpecimen.setStoreTime(tsSpecimenCheckin.getCheckinTime());
            tsSpecimen.setGetTime(tsSpecimenCheckin.getCheckinTime());
            iTestDataSampleSevice.addOrUpdateTestDataSample(tsSpecimen, testData, aplPlanLogs);
        }
    }
    
    // 8. 核心逻辑处理：根据测试项目类型分发
    List<String> ids = handleTestLogic(testData, testDataCompressor, aplContractPlan, params);
    
    // 9. 文件上传与附件关联
    if (file != null && !file.isEmpty()) {
        NewResult<String> uploadResult = iReportAttachmentService.uploadAttachment(file);
        if (uploadResult.getCode() == NewResult.SUCCESS) {
            String attachmentPath = uploadResult.getData();
            if (attachmentPath != null) {
                for (String id : ids) {
                    iReportAttachmentService.updateAttachmentPath(id, attachmentPath);
                }
                testDataCompressor.setFilePath(attachmentPath);
                iTestDataCompressorService.addOrUpdateTestDataCompressor(testDataCompressor);
            } else {
                throw new RuntimeException("附件路径为空!");
            }
        } else {
            throw new RuntimeException("附件上传失败: " + uploadResult.getMessage());
        }
    }
    
    // 10. 计算综合结论
    Integer testDataId = testData.getTestDataID();
    if (testDataId == null) {
        throw new RuntimeException("TestData ID 获取失败");
    }

    String overallConclusion = iTestReportSummaryOfResultsService.getOverallConclusion(testDataId);
    iTestDataService.addOrUpdateTestDataSingleResult(testDataId, overallConclusion);

    return new NewResult<>("自动报告功能流程已走完~", NewResult.SUCCESS);
}
